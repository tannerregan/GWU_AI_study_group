\begin{frame}
\frametitle{Chatbot vs.\ Agentic Tool}

\begin{itemize}
    \item \textbf{Chatbot} (Claude.ai, ChatGPT): you ask $\to$ it answers $\to$ \textit{you} act
    \vspace{.05in}
    \item \textbf{Agentic tool} (Claude Code): you give a goal $\to$ it reads files, writes code, runs it, fixes errors, iterates $\to$ reports back
    \vspace{.1in}
    \item For economists: instead of ``here is some code,'' it runs the code and tells you what it found
\end{itemize}

\end{frame}

\begin{frame}
\frametitle{Getting the Most Out of Your Subscription}

\begin{itemize}
    \item Subscriptions have rolling limits --- fewer tokens per task means more use
    \vspace{.05in}
    \item How to use fewer tokens:
    	\begin{itemize}
            \item[$\circ$] Start a new session for each new task: context accumulates and raises costs
            \item[$\circ$] Store standing instructions once so you never repeat yourself
            \item[$\circ$] Be specific: vague prompts waste tokens and invite unfocused replies
            \item[$\circ$] Ask multiple things at once so less back and forth
             \item[$\circ$] Compact your chats/context window
        \end{itemize}	
    \vspace{.05in}
    \item Silly trick: start Claude when wake up so 5-hour window ends earlier in day
\end{itemize}

\end{frame}

        \begin{itemize}
            \item[$\circ$] Agent lives inside your editor --- see code and AI side by side

        \end{itemize}

\section{\textcolor{violet}{Contex Engineering with Claude Overview \vspace{0.3in}}}


%%%%%%%%%%%%%%
\begin{frame}
\frametitle{Contex Engineering}

\begin{itemize}
    \item Context Window
    \begin{itemize}
        \item Each prompt to Claude Code bundles the entire history of your session
            \begin{itemize}
                \item your prompts, Claude’s responses, files it read, code outputs, system promtsz
            \end{itemize}
        \item These are all appended until the context window hits it's fixed size limit
        \begin{itemize}
            \item For Claude Code this is around 200k tokens
        \end{itemize}
        \item Claude then ``{\bf auto-compacts}'' writes it's own summary and then restarts fresh        
        \begin{itemize}
            \item less preferable to frequent intentional compaction
        \end{itemize}
    \end{itemize}
    \vspace{.05in}
    \item Contex Engineering
    \begin{itemize}
        \item You can tell claude when to compact using \hlcode{/compact your instructions}
        %\begin{itemize}
            %\item e.g. write out a new data cleaning script, compact, and ask Claude to edit and comment
        %\end{itemize}
        \item Develop habits that the context window focussed on relevant information
        \begin{itemize}
        \item Claude.md vs skills ---and--- `ask before edits' vs `plan' Modes 
        \end{itemize}
    \end{itemize}
    \vspace{.05in}
    \end{itemize}
    \vfill\footnotesize
    Read more at \href{https://paulgp.substack.com/i/192433164/what-is-a-context-window-and-how-does-it-work}{Paul GP's Guide} 
\end{frame}

\begin{frame}
\frametitle{CLAUDE.md and Skills}

\begin{itemize}
    \item \textbf{CLAUDE.md} (memory) --- Claude reads at the start of every session
        \begin{itemize}
            \item[$\circ$] Store context that is always true so you don't repeat yourself in every prompt
            \item[$\circ$] \hlcode{.claude/CLAUDE.md} is global, \hlcode{your-project/CLAUDE.md}  is {\it project} specific
            \item[$\circ$] Ask Claude to update after substantial changes to project
        \end{itemize}
    \vspace{.05in}
    \item \textbf{Skills} --- Claude reads every time it is called
        \begin{itemize}
            \item[$\circ$] Store context for a task that you repeat often, but not always relevant
            \item[$\circ$] \hlcode{.claude/skills/your-skill/SKILL.md} is called with \hlcode{/your-skill}
            \item[$\circ$] Write and update your own skills or download from others
        \end{itemize}
    \vspace{.05in}
    \end{itemize}

\begin{frame}{Files and Data that Claude is good at reading}
\begin{itemize}
    \item Claude Code works best with plain text files
    \begin{itemize}
        \item e.g. \texttt{.txt}, \texttt{.do}, \texttt{.py}, \texttt{.tex}, \texttt{.csv}, \texttt{.log}, \texttt{.md}, etc.
    \end{itemize}
    \item Tips for optimial performance with other file types
    \begin{itemize}
        \item \texttt{.docx}: convert to markdown with \hlcode{pandoc file.docx -o file.md}
        \item \texttt{.xlsx}, \texttt{.dta}, \texttt{.rds}, \texttt{.sav}: convert to CSV first with a short script
        \item \texttt{.pdf}:  
        \begin{itemize}
            \item Use the \texttt{.tex} file whenever possible (especially if equations and tables)
            \item Long PDFs (50+ pages): convert to markdown first
            \item Scanned PDFs: run OCR first, then convert to markdown
        \end{itemize}
        \end{itemize}
    \item Create a skill that loads file types, e.g. \hlcode{/pdf-to-markdown}

\section{\textcolor{violet}{Git: Version Control for Coding Projects \vspace{0.3in}}}
%%%%%%%%%%%%%%
\begin{frame}
\frametitle{Why Git Matters for Research}

\begin{itemize}
    \item Version control = a complete history of every change to your code
    \begin{itemize}
        \item No more: \texttt{analysis\_v3\_FINAL\_revised2.do}
    \end{itemize}
    \item Enables:
        \begin{itemize}
            \item[$\circ$] Backup: retrieve any prior version of your code
            \item[$\circ$] Collaborate: coauthors and RAs edit without overwriting each other
            \item[$\circ$] Track: what changed and who made the changes at every stage
        \end{itemize}
    \item Can help improve project replicability both internally and externally 
        \begin{itemize}
            \item an automatic audit trail
            \item sharing and iterating on projects publicly
        \end{itemize}
\end{itemize}
\vfill
\footnotesize Read more at Lu\'{i}s Fonseca's \href{https://luispfonseca.com/files/slides\_git.pdf}{Version Control with git (for economists)}
\end{frame}


%%%%%%%%%%%%%%
\begin{frame}
\frametitle{Git and Github Concepts}

\begin{itemize}
    \item Key concepts to remember
    \begin{itemize}
        \item \textbf{Repository (repo)}: a folder whose history Git is tracking
        \item \textbf{Commit}: a snapshot of the code at a point in time, with a descriptive message
        \item \textbf{Push / Pull / Sync}: sync your local repo with the online/remote copy
    \end{itemize}\vfill
    \item Other concepts, less critical to understand at the start:
    \begin{itemize}
        \item \textbf{Branch}: a parallel version of the code for exploration
        \item \textbf{Merge}: fold a branch back into the main code
        \item \textbf{Clone}: copy your online repo to local computer for the first time
        \item \textbf{Fork}: copy someone else's online repo to your own GitHub account
    \end{itemize}
\end{itemize}
\end{frame}


%%%%%%%%%%%%%%
\begin{frame}
\frametitle{Git vs.\ GitHub: Two Separate Things}

\begin{columns}[T]
\begin{column}{0.48\textwidth}
    \begin{center}
        \includegraphics[width=0.4\textwidth]{../figures/screenshots/git_logo.png}
    \end{center}
    \textbf{git} \textemdash{} the software that runs locally
    \begin{itemize}
        \item All history lives in the \texttt{.git} folder
        \item Records \textit{commits}, \textit{branches}, \textit{merges}
        \item Compares any two versions line-by-line; flags conflicts
    \end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
    \begin{center}
        \includegraphics[width=0.4\textwidth]{../figures/screenshots/github_logo.png}
    \end{center}
    \textbf{GitHub} \textemdash{} online hosting platform
    \begin{itemize}
        \item Stores a remote copy of your repo
        \item Handles \textit{Push}, \textit{Pull}, \textit{Clone}, \textit{Fork}
        \item Free unlimited private repos
        \item Alternatives: GitLab, Bitbucket
    \end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}
\frametitle{Typical Git workflows for an economist}
\begin{itemize}
    \item Typical workflow for an economist:
    \begin{enumerate}
        \item Pull/Sync code at start of work session (get latest version of online repo)
        \item Work on code; when something works, {\it commit} it with a descriptive message
        \item Push/Sync at end of work session (updates the online repo)
    \end{enumerate}\vfill
    \item More elaborate, with RA review
    \begin{enumerate}
        \item RA pulls repo,creates a branch from the latest commit
        \item RA works on branch, makes commits and pushes it when finished
        \item You compare the branch to the main branch
        \begin{itemize}
            \item recall git highlights all the differences line by line and flags conflicts
        \end{itemize}
        \item You merge back into main branch, or ask RA for fixes 
        \item Push/Sync to update the online repo
    \end{enumerate} 
\end{itemize}
\end{frame}
