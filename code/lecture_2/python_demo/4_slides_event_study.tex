\section{Event Study}
% ==================================================

% --------------------------------------------------
% Estimating equation
% --------------------------------------------------
\begin{frame}{Event study — specification}

  \begin{block}{Estimating equation}
    \begin{align*}
      y_{it} \;=\; \alpha_i + \gamma_t
        + \sum_{t \,\neq\, 2020} \beta_t
            \cdot \Bigl(D_i \times \mathbf{1}[t]\Bigr)
        + \varepsilon_{it}
    \end{align*}
  \end{block}

  \vspace{0.5em}

  \begin{itemize}
    \item $i$ = pixel centroid;\quad $t$ = year (2017--2025)
    \item $y_{it}$ = propensity score from \textbf{Google Satellite Embedding} for pixel $i$ in year $t$
      \begin{itemize}
        \item weights estimated using 2020 embeddings, applied to all years
      \end{itemize}
    \item $\alpha_i$ = pixel fixed effect;\quad $\gamma_t$ = year fixed effect
    \item $D_i = 1$ if pixel centroid lies \textbf{inside} the project boundary, else 0
    \item $\beta_t$: DiD coefficient for year $t$ relative to baseline 2020 \;($\beta_{2020} \equiv 0$)
    \item Standard errors clustered by 100\,m distance bin
  \end{itemize}

\end{frame}

\section{Spatial gradient}
% ==================================================

% --------------------------------------------------
% Estimating equation
% --------------------------------------------------
\begin{frame}{Spatial gradient — specification}

  \begin{block}{Estimating equation}
    \begin{align*}
      y_{it} \;=\; \alpha_i
        + \sum_{d}\;\sum_{t \,\neq\, 2020} \beta_{d,t}
            \cdot \Bigl(\mathbf{1}[\mathrm{bin}_i = d]
                        \times \mathbf{1}[t]\Bigr)
        + \varepsilon_{it}
      \qquad
      \tilde{\beta}_{d,t} \;=\; \beta_{d,t} - \beta_{d_{\max},\,t}
    \end{align*}
  \end{block}

  \vspace{0.5em}

  \begin{itemize}
    \item $i$ = pixel centroid;\quad $t$ = year (2017--2025)
    \item $y_{it}$ = propensity score from \textbf{Google Satellite Embedding} for pixel $i$ in year $t$
      \begin{itemize}
        \item probit model fitted on 2020 embeddings, applied to all years
      \end{itemize}
    \item $\alpha_i$ = pixel fixed effect;\quad year 2020 omitted for identification ($\beta_{d,2020} \equiv 0$)
    \item $\mathrm{bin}_i$ = signed 100\,m distance bin (negative = inside project boundary)
    \item $\tilde{\beta}_{d,t}$: re-centred so each year's line passes $y=0$ at the outermost bin ($d_{\max}$)
    \item Standard errors clustered by 100\,m distance bin
  \end{itemize}

\end{frame}
